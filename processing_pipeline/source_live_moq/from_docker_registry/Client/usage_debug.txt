# Ou sans mp4box :
python3 - <<'EOF'
import sys
from pathlib import Path

def dump_boxes(path):
    data = Path(path).read_bytes()
    offset = 0
    print(f"\n=== {path} ===")
    while offset < len(data) - 8:
        size = int.from_bytes(data[offset:offset+4], 'big')
        typ  = data[offset+4:offset+8].decode(errors='replace')
        print(f"  [{offset:6d}] {typ}  ({size} bytes)")
        if size < 8: break
        offset += size

for f in sorted(Path("/var/cache/nginx/segments/video").glob("seg*.m4s")):
    dump_boxes(f)
EOF



docker exec moq_client python3 - <<'EOF'
from pathlib import Path

def dump_boxes(path):
    data = Path(path).read_bytes()
    offset = 0
    print(f"\n=== {path.name} ({len(data)} bytes) ===")
    while offset < len(data) - 8:
        size = int.from_bytes(data[offset:offset+4], 'big')
        typ  = data[offset+4:offset+8].decode(errors='replace')
        print(f"  [{offset:6d}] {typ}  ({size} bytes)")
        if size < 8: break
        offset += size

segs = sorted(Path("/var/cache/nginx/segments/video").glob("seg*.m4s"))
for s in segs[:4]:
    dump_boxes(s)

print("\n=== init.mp4 ===")
init = Path("/var/cache/nginx/segments/video/init.mp4")
if init.exists():
    dump_boxes(init)
EOF
